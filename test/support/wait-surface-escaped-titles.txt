// #318 fixture, not a test. `browser-diagnostics.test.ts` copies this into a
// throwaway tree so the status classifier can be driven over titles whose SOURCE
// spelling differs from their RUNTIME name.
//
// Kept separate from `wait-surface-sample.txt` on purpose: that sample is the
// coverage guard's gap fixture and its block counts are asserted elsewhere.
// Adding blocks to it would have made this ticket edit a test file it does not
// own. Stored as `.txt` for the same reason the other sample is — its playwright
// import would otherwise make the file that reads it look like a browser test to
// the survey it is checking.
import test from "node:test";
import { chromium } from "playwright";

// An escaped quote, a unicode escape, and a hex escape in one title. At runtime
// node names this block:
//   a wired block with an escaped 'quote', a \u00e9, and a A (#318)
test("a wired block with an escaped \'quote\', a \u00e9, and a \x41 (#318)", async () => {
  const browser = await chromium.launch();
  let diagnostics: ReturnType<typeof recordBrowserDiagnostics> | null = null;
  try {
    const page = await browser.newPage();
    diagnostics = recordBrowserDiagnostics(page, page.context());
    await page.waitForSelector("#ready");
  } catch (error) {
    await captureBrowserFailure(diagnostics, "a-wired-block-with-an-escaped-title", error);
    throw error;
  } finally {
    await browser.close();
  }
});

// A second literal kind. A no-substitution template is a different AST node from
